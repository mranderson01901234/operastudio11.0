## 1. Overview

This document provides a complete implementation plan for adding image editing capabilities to OperaStudio using MCP (Model Context Protocol). Images generated by Imagen 4 will be editable via MCP tools powered by Sharp.

### Key Decisions

- ✅ **MCP Architecture**: Follows existing filesystem MCP server pattern
- ✅ **Sharp Library**: High-performance image processing
- ✅ **Database Storage**: Images stored server-side for MCP access
- ✅ **Separate MCP Server**: Dedicated image editing MCP server process
- ✅ **Auto Tool Discovery**: MCP protocol provides automatic tool discovery

### Benefits of MCP Approach

- ✅ **Auto Tool Discovery**: LLM automatically discovers available tools via `ListToolsRequestSchema`
- ✅ **Consistent Pattern**: Matches existing filesystem MCP server architecture
- ✅ **Separation of Concerns**: Image editing isolated in dedicated server
- ✅ **Extensible**: Easy to add new image editing tools
- ✅ **Type Safety**: Full TypeScript support

## 2. Architecture Design

```
    ↓

    ↓
User: "Crop this image to focus on the face"

    ↓

    ↓
Updated in imagen-context + database
```

## 3. Image Storage Strategy

### Problem Statement

**Current State:**
- Images stored in React context (client-side, in-memory)
- Lost on page refresh
- Not accessible from server-side MCP server

**Solution:**
- Store images in PostgreSQL database
- Server-side accessible for MCP tools
- Persistent across sessions
- Supports image history and editing

### Database Schema

**File:** `prisma/schema.prisma`


### Image Resolver API

**File:** `app/api/imagen/resolve/[imageId]/route.ts`

```typescript
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@clerk/nextjs/server";
import { prisma } from "@/lib/prisma";

/**
 * GET /api/imagen/resolve/[imageId]
 * 
 * Resolves imageId to actual image data for MCP server
 * Supports: "current", "img_123", "history[0]"
 */
export async function GET(
  request: NextRequest,
  { params }: { params: { imageId: string } }
) {
  try {
    const { userId } = await auth();
    if (!userId) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }

    const { imageId } = params;

    let image;

    // Resolve "current" → most recent image
    if (imageId === "current") {
      image = await prisma.generatedImage.findFirst({
        where: { userId },
        orderBy: { createdAt: "desc" },
      });
    }
    // Resolve "history[0]" → first in history, "history[1]" → second, etc.
    else if (imageId.startsWith("history[")) {
      const index = parseInt(imageId.match(/\[(\d+)\]/)?.[1] || "0");
      const images = await prisma.generatedImage.findMany({
        where: { userId },
        orderBy: { createdAt: "desc" },
        take: index + 1,
      });
      image = images[index] || null;
    }
    // Resolve by ID
    else {
      image = await prisma.generatedImage.findFirst({
        where: {
          id: imageId,
          userId, // Security: ensure user owns the image
        },
      });
    }

    if (!image) {
      return NextResponse.json(
        { error: `Image not found: ${imageId}` },
        { status: 404 }
      );
    }

    // Return image data
    return NextResponse.json({
      id: image.id,
      data: image.data,
      mimeType: image.mimeType,
      prompt: image.prompt,
      model: image.model,
      aspectRatio: image.aspectRatio,
      createdAt: image.createdAt.toISOString(),
    });
  } catch (error) {
    console.error("[Image Resolver] Error:", error);
    return NextResponse.json(
      { error: "Failed to resolve image" },
      { status: 500 }
    );
  }
}
```

```typescript
/**
 * Resolves imageId to actual image data by calling the resolver API
 */
export class ImageResolver {
  private baseUrl: string;

  constructor() {
    // Get API URL from environment or use default
    this.baseUrl =
      process.env.OPERASTUDIO_API_URL || "http://localhost:3000";
  }

  /**
   * Resolve imageId to image data
   * Supports: "current", "img_123", "history[0]"
   */
  async resolve(imageId: string): Promise<{
    id: string;
    data: string;
    mimeType: string;
    prompt: string;
    model: string;
    aspectRatio: string;
  }> {
    // Get userId from environment (set by MCP start route)
    const userId = process.env.USER_ID;
    if (!userId) {
      throw new Error("USER_ID environment variable not set");
    }

    // Call resolver API
    const url = `${this.baseUrl}/api/imagen/resolve/${encodeURIComponent(imageId)}`;
    const response = await fetch(url, {
      headers: {
        // Note: In production, you'd need to pass auth token
        // For now, we'll rely on session-based auth
        "Content-Type": "application/json",
      },
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({
        error: `HTTP ${response.status}`,
      }));
      throw new Error(
        `Failed to resolve image ${imageId}: ${error.error || response.statusText}`
      );
    }

    const imageData = await response.json();
    return imageData;
  }
}
```

### Image Editing Tools (Sharp)

**File:** `mcp-server-image-editing/src/tools/image-editing.ts`

```typescript
import sharp from "sharp";

export interface AdjustOptions {
  brightness?: number;
  contrast?: number;
  saturation?: number;
  hue?: number;
}

export class ImageEditingTools {
  /**
   * Crop image to specific region
   */
  async crop(
    imageBuffer: Buffer,
    x: number,
    y: number,
    width: number,
    height: number
  ): Promise<Buffer> {
    return await sharp(imageBuffer)
      .extract({ left: x, top: y, width, height })
      .toBuffer();
  }

  /**
   * Resize image
   */
  async resize(
    imageBuffer: Buffer,
    width: number,
    height: number,
    maintainAspectRatio: boolean = true
  ): Promise<Buffer> {
    const sharpInstance = sharp(imageBuffer);
    
    if (maintainAspectRatio) {
      return await sharpInstance.resize(width, height, { fit: "inside" }).toBuffer();
    } else {
      return await sharpInstance.resize(width, height, { fit: "fill" }).toBuffer();
    }
  }

  /**
   * Adjust image properties
   */
  async adjust(
    imageBuffer: Buffer,
    options: AdjustOptions
  ): Promise<Buffer> {
    let instance = sharp(imageBuffer);

    // Brightness and saturation via modulate
    if (options.brightness !== undefined || options.saturation !== undefined) {
      instance = instance.modulate({
        brightness: options.brightness ?? 1.0,
        saturation: options.saturation ?? 1.0,
        hue: options.hue ?? 0,
      });
    }

    // Contrast via linear
    if (options.contrast !== undefined && options.contrast !== 1.0) {
      const multiplier = options.contrast;
      const offset = -(128 * (multiplier - 1));
      instance = instance.linear(multiplier, offset);
    }

    return await instance.toBuffer();
  }

  /**
   * Apply filter
   */
  async applyFilter(
    imageBuffer: Buffer,
    filter: string,
    intensity: number
  ): Promise<Buffer> {
    let instance = sharp(imageBuffer);

    switch (filter) {
      case "blur":
        instance = instance.blur(intensity);
        break;
      case "sharpen":
        instance = instance.sharpen({
          sigma: intensity / 10,
          flat: 1,
          jagged: 2,
        });
        break;
      case "grayscale":
        instance = instance.greyscale();
        break;
      case "sepia":
        instance = instance.tint({ r: 112, g: 66, b: 20 });
        break;
      case "negate":
        instance = instance.negate();
        break;
      case "threshold":
        instance = instance.threshold(intensity);
        break;
      default:
        throw new Error(`Unknown filter: ${filter}`);
    }

    return await instance.toBuffer();
  }

  /**
   * Rotate or flip image
   */
  async rotate(
    imageBuffer: Buffer,
    angle?: number,
    flip?: string
  ): Promise<Buffer> {
    let instance = sharp(imageBuffer);

    if (flip) {
      if (flip === "horizontal" || flip === "both") {
        instance = instance.flop();
      }
      if (flip === "vertical" || flip === "both") {
        instance = instance.flip();
      }
    } else if (angle !== undefined) {
      instance = instance.rotate(angle);
    }

    return await instance.toBuffer();
  }

  /**
   * Convert format
   */
  async convertFormat(
    imageBuffer: Buffer,
    format: string,
    quality: number = 80
  ): Promise<Buffer> {
    let instance = sharp(imageBuffer);

    switch (format.toLowerCase()) {
      case "png":
        return await instance.png().toBuffer();
      case "jpeg":
      case "jpg":
        return await instance.jpeg({ quality }).toBuffer();
      case "webp":
        return await instance.webp({ quality }).toBuffer();
      case "avif":
        return await instance.avif({ quality }).toBuffer();
      default:
        throw new Error(`Unsupported format: ${format}`);
    }
  }

  /**
   * Get MIME type for format
   */
  getMimeTypeForFormat(format: string): string {
    const formatMap: Record<string, string> = {
      png: "image/png",
      jpeg: "image/jpeg",
      jpg: "image/jpeg",
      webp: "image/webp",
      avif: "image/avif",
    };
    return formatMap[format.toLowerCase()] || "image/png";
  }
}
```

```typescript
/**
 * Execute an Imagen editing tool call via MCP server
 */
async function executeImagenEditToolCall(toolCall: ToolCall): Promise<ToolResult> {
  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => {
      controller.abort();
    }, 60000); // 60 second timeout

    try {
      const response = await fetch("/api/mcp/call", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          tool: toolCall.name,
          arguments: toolCall.arguments,
        }),
        signal: controller.signal,
      });

      clearTimeout(timeoutId);

      if (!response.ok) {
        const errorData = await response.json().catch(() => ({
          error: `HTTP ${response.status}: ${response.statusText}`,
        }));

        let errorMessage = errorData.error || "Image editing failed";

        if (response.status === 404) {
          errorMessage = "Image editing MCP session not found. Please start an Image Editing session first.";
        } else if (response.status === 400) {
          errorMessage = `Invalid request: ${errorMessage}`;
        }

        return {
          callId: toolCall.id,
          name: toolCall.name,
          result: null,
          error: errorMessage,
          errorCode: response.status.toString(),
        };
      }

      const result = await response.json();

      return {
        callId: toolCall.id,
        name: toolCall.name,
        result: result,
      };
    } catch (fetchError) {
      clearTimeout(timeoutId);
      if (fetchError instanceof Error && fetchError.name === "AbortError") {
        throw new Error("Image editing operation timed out after 60 seconds");
      }
      throw fetchError;
    }
  } catch (error) {
    let errorMessage = "Unknown error occurred";

    if (error instanceof TypeError && error.message.includes("fetch")) {
      errorMessage = "Network error: Unable to reach image editing API. Check your connection.";
    } else if (error instanceof Error) {
      errorMessage = error.message;
    }

    return {
      callId: toolCall.id,
      name: toolCall.name,
      result: null,
      error: errorMessage,
    };
  }
}
```

### Example: Storing Edited Image

**File:** `app/api/imagen/store/route.ts` (NEW)

```typescript
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@clerk/nextjs/server";
import { prisma } from "@/lib/prisma";

/**
 * POST /api/imagen/store
 * Store edited image in database
 */
export async function POST(request: NextRequest) {
  try {
    const { userId } = await auth();
    if (!userId) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }

    const { imageData, mimeType, originalImageId, editType, editPrompt } =
      await request.json();

    const storedImage = await prisma.generatedImage.create({
      data: {
        userId,
        data: imageData,
        mimeType,
        originalImageId,
        editType,
        editPrompt,
        prompt: "Edited image", // Placeholder
        model: "edited",
        aspectRatio: "1:1",
      },
    });

    return NextResponse.json({
      success: true,
      image: {
        id: storedImage.id,
        data: storedImage.data,
        mimeType: storedImage.mimeType,
      },
    });
  } catch (error) {
    console.error("[Store Image] Error:", error);
    return NextResponse.json(
      { error: "Failed to store image" },
      { status: 500 }
    );
  }
}
```
